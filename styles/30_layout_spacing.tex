% ------------------------------------------------------------
% GRUNDGRÖSSE DES SATZES — die dritte Platz-Stellschraube
% ------------------------------------------------------------
% Die Klassenoption (\documentclass[8pt]) setzt die Grundgrösse, und extsizes
% kennt nur seine eigene Stufenliste — 7pt steht nicht darauf. Eine ZSF, die
% enger gesetzt werden muss, hatte deshalb bisher nur den lokalen Ausweg: ein
% rohes \fontsize direkt nach \begin{document}. Das trifft \normalsize und
% sonst nichts. \footnotesize (die `dense`-Stufe), die Titel- und Balkengrössen
% und alles aus \baselineskip Abgeleitete bleiben auf der Klassenstufe stehen;
% der Satz wird danach nicht kleiner, sondern unstimmig — und zwar lautlos,
% weil jede Grösse für sich weiter korrekt aussieht.
%
% Deshalb ist die Grundgrösse eine Stellschraube wie Dichte und Zeilenhöhe:
% EINE Zahl, die den ganzen Schriftsatz mitnimmt — die Klassenstufen, die
% benannten Schriftrollen aus 50_typography_semantics und die Balkenhöhe. Die
% VERHÄLTNISSE bleiben dabei erhalten; genau das heisst »aufeinander
% abgestimmt«. Abgestimmt heisst nicht unveränderlich.
%
%   \def\ZSFBodyFontSize{7pt}   % in preamble.tex, VOR diesem \input
%
% Leer (Default) = Klassengrösse, es wird nichts angefasst.
%
% Drei Regler, drei Fragen, bewusst nicht einer: Die Grundgrösse ändert die
% Schrift, \ZSFDensityFactor den Leerraum zwischen den Blöcken,
% \ZSFLeadingFactor den Abstand zwischen den Zeilen. Wer eine Seite sparen
% muss, hat damit drei unabhängige Hebel statt eines groben.
%
% BEWUSST AUSSERHALB: \ZSFfontFooter und \ZSFfontIdentityMarker. Beide gehören
% zur SEITE und nicht zur Textspalte — die Seitenzahl am Fuss ist eine feste
% physische Grösse und schrumpft nicht mit, nur weil der Text enger gesetzt ist.
\providecommand{\ZSFBodyFontSize}{}
\makeatletter
% Die Klassengrösse als Referenz sichern, bevor irgendetwas umgeschaltet wird.
% Auch den Zeilenabstand: 50_typography_semantics leitet daraus die Rolle des
% freien Fliesstextes ab und braucht beide Masse der Basis, nicht nur eines.
\edef\ZSF@classFontSize{\f@size}
\edef\ZSF@classBaselineSkip{\f@baselineskip}
% Der Faktor, den 50_typography_semantics für seine Schriftrollen mitliest.
\def\ZSFBodyScale{1}
% Der Leer-Test läuft über den EXPANDIERTEN Inhalt, nicht über das Makro.
% Ein Vergleich der Bedeutung (\ifx…\@empty, ebenso \tl_if_empty:N) ist hier
% falsch: \providecommand erzeugt ein \long-Makro, \@empty und \c_empty_tl
% sind keines — der Test wäre immer »nicht leer« und die Schraube liefe
% ungefragt mit Faktor 0, also mit Schriftgrösse null. \edef macht daraus
% beidseitig ein gewöhnliches Makro und vergleicht, was wirklich dasteht.
\edef\ZSF@bodySizeGiven{\ZSFBodyFontSize}
\ifx\ZSF@bodySizeGiven\@empty\else
  \ExplSyntaxOn
  \xdef\ZSFBodyScale{\fp_eval:n
    { \dim_to_fp:n {\ZSFBodyFontSize} / \dim_to_fp:n {\ZSF@classFontSize pt} }}
  \ExplSyntaxOff
\fi
% Entschieden wird am FAKTOR und nicht am Leer-Zustand: Ein ausgeschriebenes
% \ZSFBodyFontSize{8pt} (= die Klassengrösse) nimmt damit denselben Weg wie
% gar keine Angabe, statt die Klassenstufen ohne Grund neu zu schreiben.
\ifdim\ZSFBodyScale pt=1pt\else
  % Die Klassenstufen einzeln nachziehen. Gelesen wird, was die Klasse
  % TATSÄCHLICH gesetzt hat, nicht eine hier wiederholte Wertetabelle — sonst
  % driften Klasse und Umrechnung auseinander, sobald jemand die Klassenoption
  % ändert. \selectfont ist dabei stillgelegt: Gebraucht werden nur die von
  % \fontsize normalisierten Werte, nicht der geladene Font.
  \newcommand{\ZSF@rescaleSize}[1]{%
    \begingroup
      \let\selectfont\relax
      #1%
      \xdef\ZSF@rsSize{\f@size}\xdef\ZSF@rsSkip{\f@baselineskip}%
    \endgroup
    \begingroup
      \@tempdima=\ZSFBodyScale\dimexpr\ZSF@rsSize pt\relax
      \@tempdimb=\ZSFBodyScale\dimexpr\ZSF@rsSkip\relax
      \edef\ZSF@rsApply{\endgroup
        \noexpand\renewcommand\noexpand#1{%
          \noexpand\@setfontsize\noexpand#1%
            {\strip@pt\@tempdima}{\strip@pt\@tempdimb}}}%
    \ZSF@rsApply}
  \ZSF@rescaleSize{\tiny}        \ZSF@rescaleSize{\scriptsize}
  \ZSF@rescaleSize{\footnotesize}\ZSF@rescaleSize{\small}
  \ZSF@rescaleSize{\normalsize}  \ZSF@rescaleSize{\large}
  \ZSF@rescaleSize{\Large}       \ZSF@rescaleSize{\LARGE}
  \ZSF@rescaleSize{\huge}        \ZSF@rescaleSize{\Huge}
  % Sofort umschalten: alles, was weiter unten aus \baselineskip abgeleitet
  % wird (die Umbruchreserven), muss die neue Grundgrösse bereits sehen.
  \normalsize
\fi
\makeatother

% Setzt das Register sofort UND merkt sich die Rechnung. \ZSFChapterDensity

% ── Display-Skips: die EINE Stelle, und sie muss wiederholbar sein ───
% Die engen Abstände um Formeln sind eine ZSF-Entscheidung. LaTeX legt sie
% aber in den Groessenbefehlen ab: \normalsize und \footnotesize setzen alle
% vier Skips auf die KLASSENWERTE zurueck. Jeder Schriftwechsel im Inhalt
% loescht damit die Entscheidung — lautlos, denn Formeln sehen auch mit den
% Klassenwerten korrekt aus, nur luftiger.
%
% Deshalb ist die Politik ein benanntes Makro und kein einmaliges \setlength:
% Wer die Groesse wechselt, ruft sie danach erneut auf. Genutzt von
% \ZSFfontContent/\ZSFfontContentDense (der `font`-Regler), vom
% \AtBeginDocument in 70_document_settings und von \ZSFChapterDensity oben.
% \jot und \arraycolsep laufen hier mit, obwohl kein Groessenbefehl sie
% zuruecksetzt: Sie gehoeren zur selben Frage — »wie viel Luft hat eine
% Formel« — und eine zweite Anwendungsstelle waere eine zweite Auslegung.
\newcommand{\ZSFApplyDisplaySkips}{%
  \setlength{\abovedisplayskip}{\ZSFspaceXS}%
  \setlength{\belowdisplayskip}{\ZSFspaceXS}%
  \setlength{\abovedisplayshortskip}{\ZSFspaceXS}%
  \setlength{\belowdisplayshortskip}{\ZSFspaceXS}%
  \setlength{\mathsurround}{\ZSFspaceXS}%
  \setlength{\jot}{\ZSFmathRowSep}%
  \setlength{\arraycolsep}{\ZSFmathColSep}%
}

% erste sichtbare Follower setzt ihn ueber \ZSFTitleBarGapConsume. Bis dahin

% Der Verbraucher stellt \parskip vor dem Inhalt wieder her und setzt nur
% positiven Leim; eine negative Gegenkorrektur gibt es an dieser Grenze nicht.
%

% \ZSFBoxBeforeSkip (styles/README → »Abstand hat genau einen Schreiber«):

% ------------------------------------------------------------
% Vertikaler Einschub INNERHALB einer Box
% ------------------------------------------------------------
% Formelnote, Trennlinie, Item-Heading, Folgerung und die Glieder einer
% Zielkette setzen alle denselben Einschub. Als Makro statt von Hand, damit der
% Abstand an der Dichte-Stellschraube hängt und der Rhythmus an einer Stelle
% änderbar bleibt.
%
% DER ABSTAND IST EIN OPTISCHER, KEIN LEIM. Das ist der ganze Punkt dieser drei
% Zeilen und der Grund, warum sie nicht `\vspace{#1}` heissen:
%
% Zwischen zwei Blöcken im vertikalen Satz liegt nicht nur die gesetzte Glue,
% sondern zusätzlich die Zeilenschaltung — TeX schiebt vor jede Box so viel
% Leim ein, dass der Grundlinienabstand \baselineskip ergibt. Der SICHTBARE
% Abstand ist damit `#1 + (\baselineskip - Höhe der folgenden Zeile)`. Solange
% alle Zeilen gleich hoch sind (Fliesstext), faellt das nicht auf. Sobald die
% Hoehen wechseln — Pillen einer Zielkette, Displaymathe, ein Bild —, wird der
% Rhythmus zum Nebenprodukt der Tinte: gemessen an der Zielkette der Showcase
% lagen zwischen benannt gleichen Elementen 4,1 bis 11,3 pt, obwohl der Autor
% genau zwei Werte gesetzt hatte.
%
% \nointerlineskip nimmt die Zeilenschaltung für den naechsten Block heraus.
% Danach ist der Abstand von der UNTERKANTE des vorigen Blocks zur OBERKANTE
% des naechsten genau #1 — unabhaengig davon, was in den beiden steht. Weil
% Hoehe und Tiefe einer Zeile der Tinte folgen (Oberlaenge, Unterlaenge), ist
% das zugleich der optische Abstand.
%
% Das `-\parskip` bleibt aus demselben Grund wie zuvor: Der folgende Block ist
% ein Absatz, und TeX setzt \parskip beim Absatzbeginn selbst. Ohne den Abzug
% käme er zu #1 hinzu.
%
% ES MUSS \glueexpr SEIN UND NICHT \dimexpr: \parskip ist eine GLUE
% (3.5pt plus 1pt minus 0.5pt), keine Laenge. \dimexpr liest nur den
% Sollwert, der Follower zahlt beim Absatzbeginn aber die ganze Glue zurueck
% — unter dem Block blieb `#1 plus 1pt minus 0.5pt` stehen. In einem
% multicols*-Dokument ist genau dieser Leim der, mit dem TeX die Spalten
% fuellt: Der Abstand war dann das Ergebnis des Spaltenausgleichs statt #1.
% Mit \glueexpr heben sich die negativen Streckanteile gegen \parskip auf
% und uebrig bleibt starr #1 — dieselbe Rechnung wie \ZSF@blockBefore
% (60_boxes) und tcolorbox' eigenes `before skip`.
%
% Die erste Zeile faengt den Fall ab, dass zwei Einschuebe direkt aufeinander
% treffen (\ZSFsep, unmittelbar gefolgt vom naechsten Kettenglied): Dann gilt
% der zweite, statt dass sich beide addieren.
\newcommand{\ZSFInterlude}[1]{%
  \par
  \ifdim\lastskip>0pt\relax \vskip-\lastskip \fi
  \vskip\glueexpr#1-\parskip\relax
  \nointerlineskip
}
